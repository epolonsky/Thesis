# quality check with fastqc of raw reads
fastqc F7_PP_DNA_S15_R1_001.fastq.gz F7_PP_DNA_S15_R2_001.fastq.gz

# trim barcodes at first 20 base pairs. Xms and Xmx flags used to increase java memory 
trimmomatic PE -threads 2 -Xms1024m -Xmx8g -phred33 F7_PP_DNA_S15_R1_001.fastq.gz F7_PP_DNA_S15_R2_001.fastq.gz R1_paired.fastq.gz R1_unpaired.fastq.gz R2_paired.fastq.gz R2_unpaired.fastq.gz HEADCROP:20 

# quality check with fastqc of trimmed reads
fastqc R1_paired.fastq.gz  R2_paired.fastq.gz

# map trimmed reads to reference genome
bowtie2-build -f GCA_963924435.1_idCulPipi1.1_genomic.fna ref_genome_index
bowtie2 -x ref_genome_index -1 R1_paired.fastq.gz -2 R2_paired.fastq.gz --un-conc unmapped_reads.fastq.gz -S aligned.sam -p 4 > bowtie2.log 2>&1

# sam file to fasta file for blast
samtools view -h -f 4 aligned.sam > unaligned.sam
samtools fastq unaligned.sam > unaligned.fastq
seqtk seq -a unaligned.fastq > unaligned.fasta

# blast to see what the sequences that were not aligned to the reference genome are
blastn -db nt -query unaligned.fasta -out unaligned_blast.out

# sam file to sorted and indexed bam file
samtools view -bS aligned.sam | samtools sort -o aligned_sorted.bam
samtools index aligned_sorted.bam

# get coverage from samtools (shows each gene and scaffold and how much of it is covered/depth)
samtools coverage aligned_sorted.bam > aligned_sorted_coverage_output.txt

# get fasta assembly from bowtie 2 aligned file
samtools consensus -f fasta aligned_sorted.bam -o consensus.fasta 

# check assembly completeness with busco
busco -i consensus.fasta -l diptera_odb10 -o busco_out -m genome --cpu 8

# make snailplot of consensus fasta
blobtools create --fasta ~data/consensus.fasta my_snail_dataset
blobtools add --busco full_table.tsv my_snail_dataset
blobtools view --view snail --plot --out ./ snail_plot

jellyfish count -C -m 21 -s 500M -t 20 -o F7_k21_trimmed.jf <(zcat R1_paired.fastq.gz) <(zcat R2_paired.fastq.gz) > jellyfish.log 2>&1 &

# get histogram from jellyfish output
jellyfish histo -t 20 F7_k21_trimmed.jf > F7_k21_trimmed.histo

# filter genomescope with lambda 28.9 and (only accepts whole integers)

Generate a pileup file describing the aligned reads at each genomic position.

```bash
gatk Pileup -R GCA_963924435.1_idCulPipi1.1_genomic.fna -I aligned_sorted_RG.bam -O pileup_out.txt
```

# done in the data_2 directory
cd ~/data_2

# trimming with trimmomatic
mkdir trimmomatic_output
trimmomatic PE -threads 16 -Xms1024m -Xmx8g -phred33 ~/data/F7_PP_DNA_S15_R1_001.fastq.gz ~/data/F7_PP_DNA_S15_R2_001.fastq.gz trimmomatic_output/R1_trimmed_paired.fastq.gz trimmomatic_output/R1_trimmed_unpaired.fastq.gz trimmomatic_output/R2_trimmed_paired.fastq.gz trimmomatic_output/R2_trimmed_unpaired.fastq.gz HEADCROP:20 CROP:124 SLIDINGWINDOW:4:20 MINLEN:50

# quality check with fastqc of trimmed reads
mkdir fastqc_output
fastqc -o fastqc_output trimmomatic_output/R1_trimmed_paired.fastq.gz  trimmomatic_output/R2_trimmed_paired.fastq.gz

# kmer counts using jellyfish
mkdir jellyfish_output
jellyfish count -C -m 21 -s 500M -t 20 -o jellyfish_output/F7_k21_trimmed.jf <(zcat trimmomatic_output/R1_trimmed_paired.fastq.gz) <(zcat trimmomatic_output/R2_trimmed_paired.fastq.gz) > jellyfish_output/jellyfish.log 2>&1 

# kmer histogram
jellyfish histo -t 20 jellyfish_output/F7_k21_trimmed.jf > jellyfish_output/F7_k21_trimmed.histo

# genomescope 2 genome survey
mkdir genomescope_output
genomescope2 -i jellyfish_output/F7_k21_trimmed.histo -o genomescope_output -k 21 -p 2 -m 1000000

# make reference genome index for bowtie2
mkdir bowtie2_index
bowtie2-build -f ~/data/GCA_963924435.1_idCulPipi1.1_genomic.fna bowtie2_index/ref_genome_index

# map trimmed reads to reference genome
mkdir -p bowtie2_output/unmapped_reads
bowtie2 -x bowtie2_index/ref_genome_index -1 trimmomatic_output/R1_trimmed_paired.fastq.gz -2 trimmomatic_output/R2_trimmed_paired.fastq.gz --un-conc-gz bowtie2_output/unmapped_reads/unmapped_reads -S bowtie2_output/aligned.sam -p 4 > bowtie2_output/bowtie2.log 2>&1

# sort and index the aligned.bam file
samtools view -bS bowtie2_output/aligned.sam
samtools sort bowtie2_output/aligned.bam -o bowtie2_output/aligned_sorted.bam
samtools index bowtie2_output/aligned_sorted.bam

# calculate coverage using samtools coverage
samtools coverage bowtie2_output/aligned_sorted.bam > aligned_bam_coverage.txt
